cmake_minimum_required(VERSION 3.16)
project(aurora_ui_tests LANGUAGES C CXX)
get_filename_component(AURORA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL_ROOT "${AURORA_ROOT}/targets/waveshare_p4/managed_components/lvgl__lvgl")
if(NOT EXISTS "${LVGL_ROOT}/lvgl.h")
    message(FATAL_ERROR "Configure the P4 firmware first to download pinned LVGL 9.5.0")
endif()
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_ROOT}/src/*.c")
file(GLOB ASSETS "${AURORA_ROOT}/src/assets/*.c" "${AURORA_ROOT}/targets/waveshare_p4/main/assets/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES} ${ASSETS})
target_include_directories(lvgl PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${LVGL_ROOT}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_KCONFIG_IGNORE)
add_executable(ui_tests test_ui.cpp)
target_include_directories(ui_tests PRIVATE stubs "${AURORA_ROOT}/tests/native/stubs" "${AURORA_ROOT}/include" "${AURORA_ROOT}/targets/waveshare_p4/main")
target_compile_definitions(ui_tests PRIVATE AURORA_BOARD_P4 AURORA_NATIVE_TEST _CRT_SECURE_NO_WARNINGS)
target_compile_features(ui_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(lvgl PRIVATE /utf-8 /w)
    target_compile_options(ui_tests PRIVATE /utf-8 /UNDEBUG)
endif()
target_link_libraries(ui_tests PRIVATE lvgl bcrypt)
